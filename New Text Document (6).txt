function initials(name) {
  const parts = (name || "").trim().split(/\s+/).filter(Boolean);

  return (parts[0][0] + (parts[1] ? parts[1][0] : "")).toUpperCase();

function timeAgo(iso) {
  if (!iso) return "";
  const s = Math.max(1, Math.floor((Date.now() - new Date(iso).getTime()) / 1000));
  if (s < 60) return "just now";
  const m = Math.floor(s / 60); if (m < 60) return `${m}m ago`;
  const h = Math.floor(m / 60); if (h < 24) return `${h}h ago`;
  const d = Math.floor(h / 24); if (d < 7) return `${d}d ago`;
  return new Date(iso).toLocaleDateString(undefined, { month: "short", day: "numeric" });
}
function greeting() {
  const h = new Date().getHours();
  if (h < 12) return "Good morning";
  if (h < 18) return "Good afternoon";
  return "Good evening";
}
function useLocalPref(key, initial) {
  const [v, setV] = useState(() => { try { const s = localStorage.getItem(key); return s === null ? initial : JSON.parse(s); } catch (e) { return initial; } });
  const set = useCallback((next) => { setV(next); try { localStorage.setItem(key, JSON.stringify(next)); } catch (e) {} }, [key]);
  return [v, set];
}
// Keeps a closing element mounted for `delay` ms after it goes inactive,
// so its CSS exit animation/transition has time to finish instead of the
// element just vanishing mid-motion.
function useDelayedMount(active, delay) {
  const [mounted, setMounted] = useState(active);
  useEffect(() => {
    if (active) { setMounted(true); return; }
    const t = setTimeout(() => setMounted(false), delay);
    return () => clearTimeout(t);
  }, [active, delay]);
  return mounted;
}

}

/* ── Sheet: bottom sheet (mobile) / centered panel (desktop), with
   pointer-tracked, 1:1, interruptible drag-to-dismiss on the handle.
   The drag itself carries no CSS transition — it must stay glued to
   the finger; a transition is added only for the release settle. ── */
function Sheet({ onClose, children, labelledBy }) {
  const sheetRef = useRef(null);
  const drag = useRef({ active: false, startY: 0, lastY: 0, lastT: 0, v: 0 });
  const [closing, setClosing] = useState(false);

  const requestClose = useCallback(() => {
    setClosing(true);
    setTimeout(onClose, 180);
  }, [onClose]);

  useEffect(() => {
    const onKey = (e) => { if (e.key === "Escape") requestClose(); };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [requestClose]);

  function onPointerDown(e) {
    if (window.innerWidth >= 640) return;
    const el = sheetRef.current; if (!el) return;
    el.style.transition = "none";
    el.setPointerCapture(e.pointerId);
    drag.current = { active: true, startY: e.clientY, lastY: e.clientY, lastT: performance.now(), v: 0 };
  }
  function onPointerMove(e) {
    if (!drag.current.active) return;
    const el = sheetRef.current; if (!el) return;
    let dy = e.clientY - drag.current.startY;
    if (dy < 0) dy = dy / 3;
    el.style.transform = `translateY(${dy}px)`;
    const now = performance.now(), dt = Math.max(1, now - drag.current.lastT);
    drag.current.v = (e.clientY - drag.current.lastY) / dt;
    drag.current.lastY = e.clientY; drag.current.lastT = now; drag.current.last = dy;
  }
  function onPointerUp() {
    if (!drag.current.active) return;
    drag.current.active = false;
    const el = sheetRef.current; if (!el) return;
    const dy = drag.current.last || 0, tossing = drag.current.v > 0.5;
    el.style.transition = "transform .28s cubic-bezier(.2,.8,.2,1)";
    if (dy > 110 || tossing) { el.style.transform = `translateY(100%)`; requestClose(); }
    else { el.style.transform = "translateY(0)"; }
  }

  return <div className="sheet-backdrop" onClick={requestClose} style={closing ? { animation: "fadeIn .18s ease-in reverse" } : undefined}>
    <div ref={sheetRef} className="sheet" role="dialog" aria-modal="true" aria-labelledby={labelledBy}
      onClick={(e) => e.stopPropagation()}
      style={closing ? { transition: "transform .18s ease-in, opacity .18s ease-in", transform: "translateY(24px)", opacity: 0 } : undefined}>
      <div className="sheet-handle" onPointerDown={onPointerDown} onPointerMove={onPointerMove} onPointerUp={onPointerUp} onPointerCancel={onPointerUp} style={{ cursor: "grab", touchAction: "none" }} />
      {children}
    </div>
  </div>;
}

      {matchedProjects.length > 0 && <>
        <div className="section-head"><h3 className="t-title">Projects</h3></div>
        <div className="grid">{matchedProjects.map((p, i) => <ProjectCardLite key={p.id} p={p} index={i} onOpen={onOpenProject} count={items.filter(it => it.projectId === p.id).length} />)}</div>
      </>}
      {matchedItems.length > 0 && <>
        <div className="section-head"><h3 className="t-title">Items</h3></div>
        <div className="grid">{matchedItems.map((it, i) => <ItemCard key={it.id} it={it} index={i} project={projectsById[it.projectId]} onOpen={onOpenItem} />)}</div>
      </>}
      {matchedProjects.length === 0 && matchedItems.length === 0 && <div className="empty" style={{ marginTop: "var(--sp-8)" }}>
        <Icon name="search" size={30} style={{ opacity: .35 }} /><p className="t-body" style={{ marginTop: 10 }}>Nothing matches "{query}".</p>
      </div>}
    </div>;
  }

  const recentProjects = projects.slice(0, 6);
  const recentItems = items.slice(0, 6);

  return <div className="page-in">
    <h1 className="t-display">{greeting()}</h1>
    <p className="t-body" style={{ color: "var(--muted-foreground)", marginTop: 4 }}>
      {projects.length} project{projects.length === 1 ? "" : "s"} · {items.length} item{items.length === 1 ? "" : "s"} across your library.
    </p>

    <div className="stat-grid">
      {LIBRARY_NAV.map((n, i) => <StatCard key={n.key} icon={n.icon} label={n.label} count={counts[n.key] || 0} index={i} onClick={() => onSelectView(n.key)} />)}
    </div>

    <div className="section-head">
      <h3 className="t-title">Recent projects</h3>
      {projects.length > 6 && <button className="see-all" onClick={() => onSelectView("projects")}>See all <Icon name="chevronRight" size={14} /></button>}
    </div>
    {recentProjects.length === 0
      ? <div className="empty"><Icon name="folder" size={28} style={{ opacity: .35 }} /><p className="t-body" style={{ marginTop: 10 }}>No projects yet — create your first one.</p></div>
      : <div className="grid">{recentProjects.map((p, i) => <ProjectCardLite key={p.id} p={p} index={i} onOpen={onOpenProject} count={items.filter(it => it.projectId === p.id).length} />)}</div>}

    <div className="section-head">
      <h3 className="t-title">Recent items</h3>
    </div>
    {recentItems.length === 0
      ? <div className="empty"><Icon name="fileText" size={26} style={{ opacity: .35 }} /><p className="t-body" style={{ marginTop: 10 }}>Items you add will show up here.</p></div>
      : <div style={{ display: "flex", flexDirection: "column", gap: 4 }}>{recentItems.map((it, i) => <ItemRow key={it.id} it={it} index={i} project={projectsById[it.projectId]} onOpen={onOpenItem} />)}</div>}
  </div>;
}
/* lightweight variant of ProjectCard without the kebab menu, for Overview/search */
function ProjectCardLite({ p, count, index, onOpen }) {
  return <button className="card proj-card proj-card-btn stagger" style={{ animationDelay: `${index * 30}ms` }} onClick={() => onOpen(p.id)}>
    <div className="proj-glyph"><Icon name="folder" size={19} /></div>
    <div style={{ minWidth: 0, width: "100%" }}>
      <div className="t-title" style={ellipsis}>{p.title}</div>
      {p.subject && <div className="t-label" style={{ marginTop: 3 }}>{p.subject}</div>}
    </div>
    <div style={{ display: "flex", alignItems: "center", justifyContent: "space-between", width: "100%" }}>
      <span className="t-label" style={{ textTransform: "none", letterSpacing: 0, fontWeight: 500 }}>{count} item{count === 1 ? "" : "s"}</span>
      <span className="t-label" style={{ textTransform: "none", letterSpacing: 0, fontWeight: 500 }}>{timeAgo(p.createdAt)}</span>
    </div>
  </button>;
}

/* ── Projects view (full grid, search, kebab menu) ───────────── */
function ProjectsView({ query, projects, items, degraded, onOpenProject, openMenuId, onToggleMenu, onRename, onDelete, onShare }) {
  const filtered = useMemo(() => {
    const q = query.trim().toLowerCase();
    return !q ? projects : projects.filter(p => (p.title + " " + (p.subject || "")).toLowerCase().includes(q));
  }, [projects, query]);

  return <div className="page-in">
    {degraded && <div className="banner">Running without saved storage right now — projects you add will disappear on reload.</div>}
    {filtered.length === 0
      ? <div className="empty" style={{ marginTop: "var(--sp-8)" }}>
        <Icon name="folder" size={34} style={{ opacity: .35 }} />
        <p className="t-body" style={{ marginTop: 10 }}>{projects.length === 0 ? "No projects yet — add your first one." : "No projects match your search."}</p>
      </div>
      : <div className="grid">
        {filtered.map((p, i) => <ProjectCard key={p.id} p={p} index={i} count={items.filter(it => it.projectId === p.id).length}
          onOpen={onOpenProject} menuOpen={openMenuId === p.id} onToggleMenu={onToggleMenu} onRename={onRename} onDelete={onDelete} onShare={onShare} />)}
      </div>}
  </div>;
}

/* ── Library view (single item type across all projects) ─────── */
function LibraryView({ view, query, items, projectsById, onOpenItem }) {
  const filtered = useMemo(() => {
    const q = query.trim().toLowerCase();
    const scoped = items.filter(i => i.type === view);
    return !q ? scoped : scoped.filter(i => i.title.toLowerCase().includes(q));
  }, [items, view, query]);

  return <div className="page-in">
    <p className="t-body" style={{ color: "var(--muted-foreground)", margin: "0 0 4px" }}>{LIBRARY_DESC[view]}</p>
    {filtered.length === 0
      ? <div className="empty" style={{ marginTop: "var(--sp-6)" }}>
        <Icon name={(LIBRARY_NAV.find(n => n.key === view) || {}).icon || "fileText"} size={32} style={{ opacity: .35 }} />
        <p className="t-body" style={{ marginTop: 10 }}>{items.some(i => i.type === view) ? "Nothing matches your search." : `No ${VIEW_TITLE[view].toLowerCase()} yet — add some from inside a project.`}</p>
      </div>
      : <div className="grid">{filtered.map((it, i) => <ItemCard key={it.id} it={it} index={i} project={projectsById[it.projectId]} onOpen={onOpenItem} />)}</div>}
  </div>;
}

  </div>;

/* ── Project detail (full-screen, replaces the shell) ─────────── */
function ProjectDetail({ project, items, onBack, onAddItem, onOpenItem, onDeleteProject, onShare }) {
  const [confirmDelete, setConfirmDelete] = useState(false);
  return <div className="page-in" style={{ minHeight: "100%" }}>
    <div className="topbar glass" style={{ display: "flex", alignItems: "center", gap: 10, borderBottomColor: "var(--glass-border)" }}>
      <IconBtn name="chevronLeft" label="Back" onClick={onBack} />
      <div style={{ flex: 1, minWidth: 0 }}>
        <div className="t-title" style={ellipsis}>{project.title}</div>
        {project.subject && <div className="t-label" style={{ marginTop: 1 }}>{project.subject}</div>}
      </div>
      <IconBtn name="share" label="Share project" onClick={() => onShare(project)} />
      <IconBtn name="trash" label="Delete project" onClick={() => setConfirmDelete(true)} />
    </div>

    <div className="container" style={{ paddingTop: "var(--sp-4)" }}>
      {items.length === 0
        ? <div className="empty"><Icon name="folder" size={30} style={{ opacity: .35 }} /><p className="t-body" style={{ marginTop: 10 }}>Nothing here yet. Add a Figma link, a note, a PDF, or your slides.</p></div>
        : <div style={{ display: "flex", flexDirection: "column", gap: 4 }}>
          {items.map((it, i) => {
            const meta = TYPE_META[it.type] || TYPE_META.note;
            return <button key={it.id} className="item-row stagger" style={{ animationDelay: `${i * 30}ms` }} onClick={() => onOpenItem(it)}>
              <div className="item-icon"><Icon name={meta.icon} size={17} /></div>
              <div style={{ flex: 1, minWidth: 0 }}>
                <div className="t-body" style={{ fontWeight: 600, ...ellipsis }}>{it.title}</div>
                <div className="t-label" style={{ fontWeight: 500, textTransform: "none", letterSpacing: 0 }}>{meta.label} · {timeAgo(it.createdAt)}</div>
              </div>
              <Icon name="chevronRight" size={16} style={{ color: "var(--muted-foreground)", flexShrink: 0 }} />
            </button>;
          })}
        </div>}
    </div>

    <button className="btn-icon filled press-lg" onClick={onAddItem} aria-label="Add item"
      style={{ position: "fixed", right: 24, bottom: 28, width: 56, height: 56, boxShadow: "var(--shadow-md)" }}>
      <Icon name="plus" size={24} />
    </button>

    {confirmDelete && <ConfirmSheet title="Delete project?" body={`"${project.title}" and everything in it will be removed. This can't be undone.`}
      onConfirm={onDeleteProject} onClose={() => setConfirmDelete(false)} />}
  </div>;
}

/* ── App ──────────────────────────────────────────────────────── */
function App() {
  const [dark, setDark] = useLocalPref("ap-dark", false);
  const [profileName, setProfileName] = useLocalPref("ap-profile-name", "You");
  const [ready, setReady] = useState(false);
  const dbRef = useRef(null), assetsRef = useRef(null);
  const [degraded, setDegraded] = useState(false);

  const [projects, setProjects] = useState([]);
  const [items, setItems] = useState([]); // ALL items, app-wide
  const [selected, setSelected] = useState(null); // project id (full-screen detail)
  const [view, setView] = useState("overview"); // overview | projects | figma | note | pdf | slides
  const [query, setQuery] = useState("");
  const [sidebarOpen, setSidebarOpen] = useState(false);
  const [openMenuId, setOpenMenuId] = useState(null);

  const [addingProject, setAddingProject] = useState(false);
  const [editingProject, setEditingProject] = useState(null); // project being renamed
  const [confirmDeleteProject, setConfirmDeleteProject] = useState(null); // project id
  const [addingItem, setAddingItem] = useState(false);
  const [viewingItem, setViewingItem] = useState(null);
  const [uploading, setUploading] = useState(false);
  const [scrolled, setScrolled] = useState(false);
  const [toast, setToast] = useState(null); // { id, text }

  const localRef = useRef({ projects: [], items: [] });
  const toastIdRef = useRef(0);
  const hashProjectHandled = useRef(false);
  const hashItemHandled = useRef(false);

  const pushToast = useCallback((text) => {
    toastIdRef.current += 1;
    setToast({ id: toastIdRef.current, text });
  }, []);

  const shareLink = useCallback(async (hash, title) => {
    const url = buildShareUrl(hash);
    try {
      if (navigator.share) { await navigator.share({ url, title }); }
      else if (navigator.clipboard && navigator.clipboard.writeText) { await navigator.clipboard.writeText(url); pushToast("Link copied"); }
      else { pushToast("Sharing isn't supported here"); }
    } catch (e) { /* user dismissed the share sheet — not an error */ }
  }, [pushToast]);
  const shareProject = useCallback((p) => shareLink(`project/${p.id}`, p.title), [shareLink]);
  const shareItem = useCallback((it) => shareLink(`item/${it.id}`, it.title), [shareLink]);
  const drawerMounted = useDelayedMount(sidebarOpen, 300);

  useEffect(() => {
    let cancelled = false;
    (async () => {
      const db = typeof window.claude !== "undefined" ? await window.claude.use("db") : null;
      const assets = typeof window.claude !== "undefined" ? await window.claude.use("assets") : null;
      if (cancelled) return;
      dbRef.current = db; assetsRef.current = assets;
      setDegraded(!db);
      setReady(true);
    })();
    return () => { cancelled = true; };
  }, []);

  // Resolves a #project/<id> or #item/<id> link once, the first time
  // that collection's data becomes available — so opening a shared
  // link lands directly on the right project or item instead of the
  // overview.
  function resolveProjectHash(list) {
    if (hashProjectHandled.current) return;
    const h = parseHash();
    if (h.type === "project" && list.some(p => p.id === h.id)) { hashProjectHandled.current = true; setSelected(h.id); }
  }
  function resolveItemHash(list) {
    if (hashItemHandled.current) return;
    const h = parseHash();
    if (h.type === "item") {
      const it = list.find(i => i.id === h.id);
      if (it) { hashItemHandled.current = true; setViewingItem(it); }
    }
  }

  useEffect(() => {
    if (!ready) return;
    const db = dbRef.current;
    if (!db) { setProjects(localRef.current.projects); resolveProjectHash(localRef.current.projects); return; }
    const unsub = db.collection("projects").orderBy("createdAt", "desc").onSnapshot(
      snap => { const list = snap.docs.map(d => ({ id: d.id, ...d.data() })); setProjects(list); resolveProjectHash(list); },
      () => {}
    );
    return () => unsub && unsub();
  }, [ready]);

  useEffect(() => {
    if (!ready) return;
    const db = dbRef.current;
    if (!db) { setItems(localRef.current.items); resolveItemHash(localRef.current.items); return; }
    const unsub = db.collection("items").orderBy("createdAt", "desc").onSnapshot(
      snap => { const list = snap.docs.map(d => ({ id: d.id, ...d.data() })); setItems(list); resolveItemHash(list); },
      () => {}
    );
    return () => unsub && unsub();
  }, [ready]);

  useEffect(() => {
    if (dark) document.documentElement.setAttribute("data-theme", "dark");
    else document.documentElement.removeAttribute("data-theme");
  }, [dark]);

  useEffect(() => {
    const onScroll = () => setScrolled(window.scrollY > 4);
    onScroll();
    window.addEventListener("scroll", onScroll);
    return () => window.removeEventListener("scroll", onScroll);
  }, []);

  useEffect(() => {
    if (!openMenuId) return;
    const close = () => setOpenMenuId(null);
    window.addEventListener("click", close);
    return () => window.removeEventListener("click", close);
  }, [openMenuId]);

  useEffect(() => {
    const onKey = (e) => { if (e.key === "Escape") { setSidebarOpen(false); setOpenMenuId(null); } };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, []);

  const project = projects.find(p => p.id === selected) || null;
  const projectItems = useMemo(() => items.filter(i => i.projectId === selected), [items, selected]);
  const projectsById = useMemo(() => Object.fromEntries(projects.map(p => [p.id, p])), [projects]);
  const counts = useMemo(() => {
    const c = { figma: 0, note: 0, pdf: 0, slides: 0 };
    items.forEach(i => { if (c[i.type] !== undefined) c[i.type]++; });
    return c;
  }, [items]);

  function now() { return new Date().toISOString(); }

  function selectView(v) { setView(v); setQuery(""); setSidebarOpen(false); }
  function openProject(id) {
    setSelected(id); setSidebarOpen(false); setViewingItem(null);
    history.replaceState(null, "", "#project/" + id);
  }
  function closeProject() {
    setSelected(null);
    history.replaceState(null, "", window.location.pathname + window.location.search);
  }

  async function createProject({ title, subject }) {
    const db = dbRef.current;
    if (db) { await db.collection("projects").add({ title, subject, createdAt: now() }); }
    else {
      const p = { id: "p" + Date.now(), title, subject, createdAt: now() };
      localRef.current.projects = [p, ...localRef.current.projects];
      setProjects([...localRef.current.projects]);
    }
    setAddingProject(false);
  }

  async function saveProjectEdit(id, patch) {
    const db = dbRef.current;
    if (db) { await db.collection("projects").doc(id).update(patch); }
    else {
      localRef.current.projects = localRef.current.projects.map(p => p.id === id ? { ...p, ...patch } : p);
      setProjects([...localRef.current.projects]);
    }
    setEditingProject(null);
  }

  function handleSaveProject(data) {
    if (data.id) saveProjectEdit(data.id, { title: data.title, subject: data.subject });
    else createProject(data);
  }

  async function deleteProjectById(id) {
    const db = dbRef.current, assets = assetsRef.current;
    if (db) {
      const snap = await db.collection("items").where("projectId", "==", id).get();
      for (const d of snap.docs) {
        const data = d.data();
        if (data.assetId && assets) { try { await assets.delete(data.assetId); } catch (e) {} }
        await d.ref.delete();
      }
      await db.collection("projects").doc(id).delete();
    } else {
      localRef.current.items = localRef.current.items.filter(i => i.projectId !== id);
      localRef.current.projects = localRef.current.projects.filter(p => p.id !== id);
      setItems([...localRef.current.items]); setProjects([...localRef.current.projects]);
    }
    if (selected === id) setSelected(null);
    setConfirmDeleteProject(null);
  }

  async function addLinkItem({ title, url, type }) {
    const rec = { projectId: selected, type, title, url, createdAt: now() };
    const db = dbRef.current;
    if (db) await db.collection("items").add(rec);
    else { const it = { id: "i" + Date.now(), ...rec }; localRef.current.items = [it, ...localRef.current.items]; setItems([...localRef.current.items]); }
    setAddingItem(false);
  }
  async function addNoteItem({ title, body }) {
    const rec = { projectId: selected, type: "note", title, body, createdAt: now() };
    const db = dbRef.current;
    if (db) await db.collection("items").add(rec);
    else { const it = { id: "i" + Date.now(), ...rec }; localRef.current.items = [it, ...localRef.current.items]; setItems([...localRef.current.items]); }
    setAddingItem(false);
  }
  async function addFileItem({ title, file }) {
    const db = dbRef.current, assets = assetsRef.current;
    if (!db || !assets) {
      const url = URL.createObjectURL(file);
      const it = { id: "i" + Date.now(), projectId: selected, type: "pdf", title, assetUrl: url, createdAt: now() };
      localRef.current.items = [it, ...localRef.current.items]; setItems([...localRef.current.items]);
      setAddingItem(false); return;
    }
    setUploading(true);
    try {
      const res = await assets.upload(file, { type: "application/pdf" });
      await db.collection("items").add({ projectId: selected, type: "pdf", title, assetId: res.id, assetUrl: res.url, sizeBytes: res.sizeBytes, createdAt: now() });
      setAddingItem(false);
    } finally { setUploading(false); }
  }
  async function deleteItem(it) {
    const db = dbRef.current, assets = assetsRef.current;
    if (db) {
      if (it.assetId && assets) { try { await assets.delete(it.assetId); } catch (e) {} }
      await db.collection("items").doc(it.id).delete();
    } else {
      localRef.current.items = localRef.current.items.filter(i => i.id !== it.id);
      setItems([...localRef.current.items]);
    }
    setViewingItem(null);
  }
  async function saveNoteBody(it, body) {
    const db = dbRef.current;
    if (db) await db.collection("items").doc(it.id).update({ body, updatedAt: now() });
    else { localRef.current.items = localRef.current.items.map(i => i.id === it.id ? { ...i, body } : i); setItems([...localRef.current.items]); }
    setViewingItem(v => v ? { ...v, body } : v);
  }

  function goToProjectFromViewer(id) { setViewingItem(null); setSelected(id); }

  if (project) {
    return <>
      <ProjectDetail project={project} items={projectItems} onBack={() => setSelected(null)}
        onAddItem={() => setAddingItem(true)} onOpenItem={setViewingItem} onDeleteProject={() => deleteProjectById(project.id)} />
      {addingItem && <AddItemSheet uploading={uploading} onClose={() => setAddingItem(false)}
        onSaveLink={addLinkItem} onSaveNote={addNoteItem} onSaveFile={addFileItem} />}
      {viewingItem && <ItemViewer item={viewingItem} project={project} hideProjectChip
        onClose={() => setViewingItem(null)} onDelete={() => deleteItem(viewingItem)} onSaveNote={(b) => saveNoteBody(viewingItem, b)} />}
    </>;
  }

  const drawerMounted = useDelayedMount(sidebarOpen, 300);

  return <div className="app-shell">
    {drawerMounted && <div className={`drawer-backdrop ${sidebarOpen ? "" : "closing"}`} onClick={() => setSidebarOpen(false)} />}
    <Sidebar open={sidebarOpen} onClose={() => setSidebarOpen(false)} view={view} onSelectView={selectView}
      projectCount={projects.length} counts={counts} profileName={profileName}
      onNewProject={() => { setAddingProject(true); setSidebarOpen(false); }} />

    <div className="main">
      <div className={`topbar2 ${scrolled ? "is-glass glass" : ""}`}>
        <div style={{ display: "flex", alignItems: "center", gap: 10, minWidth: 0 }}>
          <button className="btn-icon outline hamburger-btn" aria-label="Open menu" onClick={() => setSidebarOpen(true)}><Icon name="menu" size={17} /></button>
          <div className="topbar-brand"><Logo size={16} /></div>
          <h1 className="t-display" style={ellipsis}>{VIEW_TITLE[view]}</h1>
        </div>
        <div style={{ display: "flex", alignItems: "center", gap: 10 }}>
          {view !== "profile" && <div className="search-pill">
            <Icon name="search" size={15} style={{ color: "var(--muted-foreground)", flexShrink: 0 }} />
            <input value={query} onChange={e => setQuery(e.target.value)} placeholder={view === "overview" ? "Search everything…" : `Search ${VIEW_TITLE[view].toLowerCase()}…`} />
          </div>}
          <IconBtn name={dark ? "sun" : "moon"} label="Toggle theme" onClick={() => setDark(!dark)} />
        </div>
      </div>

      <div className="wide-container">
        {view === "overview" && <Overview query={query} projects={projects} items={items} counts={counts} projectsById={projectsById}
          onOpenProject={openProject} onOpenItem={setViewingItem} onSelectView={selectView} />}
        {view === "projects" && <ProjectsView query={query} projects={projects} items={items} degraded={degraded}
          onOpenProject={openProject} openMenuId={openMenuId}
          onToggleMenu={(id) => setOpenMenuId(cur => cur === id ? null : id)}
          onRename={(p) => { setEditingProject(p); setOpenMenuId(null); }}
          onDelete={(id) => { setConfirmDeleteProject(id); setOpenMenuId(null); }} />}
        {["figma", "note", "pdf", "slides"].includes(view) && <LibraryView view={view} query={query} items={items} projectsById={projectsById} onOpenItem={setViewingItem} />}
        {view === "profile" && <ProfilePage name={profileName} onNameChange={setProfileName} dark={dark} onToggleDark={() => setDark(!dark)}
          projectCount={projects.length} itemCount={items.length} counts={counts} degraded={degraded} />}
      </div>
    </div>

    <button className="btn-icon filled press-lg mobile-fab" onClick={() => setAddingProject(true)} aria-label="New project"
      style={{ position: "fixed", right: 24, bottom: 28, width: 56, height: 56, boxShadow: "var(--shadow-md)" }}>
      <Icon name="plus" size={24} />
    </button>

    {addingProject && <ProjectSheet onClose={() => setAddingProject(false)} onSave={handleSaveProject} />}
    {editingProject && <ProjectSheet editing={editingProject} onClose={() => setEditingProject(null)} onSave={handleSaveProject} />}
    {confirmDeleteProject && <ConfirmSheet title="Delete project?"
      body={`"${(projectsById[confirmDeleteProject] || {}).title || "This project"}" and everything in it will be removed. This can't be undone.`}
      onConfirm={() => deleteProjectById(confirmDeleteProject)} onClose={() => setConfirmDeleteProject(null)} />}
    {viewingItem && <ItemViewer item={viewingItem} project={projectsById[viewingItem.projectId]}
      onOpenProject={goToProjectFromViewer} onClose={() => setViewingItem(null)}
      onDelete={() => deleteItem(viewingItem)} onSaveNote={(b) => saveNoteBody(viewingItem, b)} />}
  </div>;
}

ReactDOM.createRoot(document.getElementById("root")).render(<App />);
</script>
